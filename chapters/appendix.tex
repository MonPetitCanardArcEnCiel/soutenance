\subsection{More on the Physical Optics solver}
\begin{frame}
    \frametitle{Discrete Pysical Optics}
    \label{fra:discrete-physical-optics}
    The skew surface $\Gamma$ is approximated by a set of triangles $(T_n)_{1\leq n\leq N}$. On each triangle, the vector $\n_n$ is constant. Finally, let us consider the monostatic observation case i.e.~$\uu_s = -\uu_i$.

    The total far field is the superposition of the contribution of each triangle :
    
    \begin{equation}
        \boxed{
        \E^{\infty}(\uu_s) = \frac{2k}{4\pi}\sum_n\left[ \uu_s\times\left(\n_n\times \uu_s\times\E\inc\right)\times \uu_s\right]\int_{T_n}e^{-2jk \uu_s\cdot y}\dd T_n(y)
        }
    \end{equation}

    \hyperlink{fra:integral-discretization}{\beamerbutton{More info}}
    % \normalsize
\end{frame}

\begin{frame}
    \frametitle{The PTD extension}
\end{frame}

\begin{frame}
    \frametitle{The discretization of the integral}
    \label{fra:integral-discretization}


    Let : $\mathcal I_n = \int_{T_n}e^{-2jk\uu_s\cdot y}\dd T_n(y).$
    \pause
    The Gordon formula\footnote<3->{Laval, D.~, 2006, \textit{Application de Méthodes Asymptotiques à la Simulation de la Diffraction \'Electromagnétique par un Corps Régulier}.} gives :%
    \begin{equation}
        \mathcal I_n=\frac{j}{2kQ_n}\sum_{m=1}^{3}(q_n\cdot a_{n,m})e^{-2jk(x_{n,m}-x_{n,0})\cdot l_s}\sin_c\left(\frac{2k l_s\cdot a_{n,m}}{2}\right),
    \end{equation}%
    \small
    with $l_s = -2 \uu_s$, $Q_n = |n_n\times l_s|$ et $q_n = \frac{\n_n\times l_s}{Q_n}$. $a_{n,m}$ is the vector of the edge number $m$ of $T_n$, and $x_{n,0}$ is the unit vector towards the barycenter of $T_n$ and $x_{n,m}$ towards the $m^{\text{th}}$ vertex of $T_n$. $\sin_c(x) = \frac{\sin(x)}{x}$.

\end{frame}

\subsection{Adjoint derivation}

\begin{frame}

    \frametitle{Continuous adjoint formulation}
    Let us rewrite the linear system of unknowns $\J$ defined by equation \eqref{eq:syslin} under the following form :
    \begin{equation}
        \mathcal R(\X(\eta),\J(\eta)) = 0,
        \label{eq:system-cont}
    \end{equation}
    and consider the observation function $\zeta(\X(\eta),\J(\eta))$.
    The right derivative of $\zeta$ with respect to $\eta$ writes :
    \begin{equation}
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} + \derpart{\zeta}{\J}\derpart{\J}{\eta}\right)\dd \eta.
        \label{eq:dsigmadx}
    \end{equation}

    \only<1>{
        $\X$ : mesh coordinates

        $\eta$ : optimization parameters
        }

    \onslide<2->
    The term $\derpart{\J}{\eta}$ is very costly to compute. However, it can be expressed in a more clever way since $\dd\RR = 0$ :

    \begin{equation}
        \dd\RR = \left(\derpart\RR\X\derpart\X \eta + \derpart\RR\J\derpart\J \eta\right)\dd \eta = 0.
    \end{equation}
    \onslide<3->
    Then :
    \begin{equation}
        \derpart\J \eta = \left(\derpart\RR\J\right)^{-1}\derpart\RR\X\derpart\X \eta.
        \label{eq:djdx}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Continuous adjoint formulation II}
    Injecting the system \eqref{eq:djdx} into the expression \eqref{eq:dsigmadx}, yields :

    \begin{equation}
        \boxed{
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} - \Psi^*\derpart{\mathcal R}{\X}\derpart{\X}{\eta}\right)\dd\eta.
        }
        \label{eq:dsigmadx-adj}
    \end{equation}

    Where $\Psi$ is the solution to the adjoint system: 
    \begin{equation}
        \left(\derpart{\mathcal R}{\J}\right)^* \Psi = \left(\derpart{\zeta}{\J}\right)^*.
        \label{eq:adjoint-cont}
    \end{equation}
\end{frame}
