\subsection{More on the Physical Optics solver}
\begin{frame}
    \frametitle{Discrete Pysical Optics}
    \label{fra:discrete-physical-optics}
    The skew surface $\Gamma$ is approximated by a set of triangles $(T_n)_{1\leq n\leq N}$. On each triangle, the vector $\n_n$ is constant. Finally, let us consider the monostatic observation case i.e.~$\uu_s = -\uu_i$.

    The total far field is the superposition of the contribution of each triangle:
    
    \begin{equation}
        \boxed{
        \E^{\infty}(\uu_s) = \frac{2k}{4\pi}\sum_n\left[ \uu_s\times\left(\n_n\times \uu_s\times\E\inc\right)\times \uu_s\right]\int_{T_n}e^{-2jk \uu_s\cdot y}\dd T_n(y)
        }
    \end{equation}

    % \normalsize
\end{frame}

\begin{frame}
    \frametitle{The discretization of the integral}
    \label{fra:integral-discretization}


    Let: $\mathcal I_n = \int_{T_n}e^{-2jk\uu_s\cdot y}\dd T_n(y).$
    \pause
    % The Gordon formula\footnote<3->{Laval, D.~, 2006, \textit{Application de Méthodes Asymptotiques à la Simulation de la Diffraction \'Electromagnétique par un Corps Régulier}.} gives:%
    The Gordon formula\footcite{Laval2006} gives:%
    \begin{equation}
        \mathcal I_n=\frac{j}{2kQ_n}\sum_{m=1}^{3}(q_n\cdot a_{n,m})e^{-2jk(x_{n,m}-x_{n,0})\cdot l_s}\sin_c\left(\frac{2k l_s\cdot a_{n,m}}{2}\right),
    \end{equation}%
    \small
    with $l_s = -2 \uu_s$, $Q_n = |n_n\times l_s|$ and $q_n = \frac{\n_n\times l_s}{Q_n}$. $a_{n,m}$ is the vector of the edge number $m$ of $T_n$, and $x_{n,0}$ is the unit vector towards the barycenter of $T_n$ and $x_{n,m}$ towards the $m^{\text{th}}$ vertex of $T_n$. $\sin_c(x) = \frac{\sin(x)}{x}$.    

    \hyperlink{fra:discrete-physical-optics}{\beamerbutton{Back}}
\end{frame}



\begin{frame}
    \frametitle{State of the art -- BEM adjoint optimization}
    \small
    Adjoint BEM formulation have been used for a while for electromagnetic optimization and sensitivity analysis
    \begin{itemize}[label=\ding{227},leftmargin=5mm]
        \item 1995: Use of Lions' optimal control theory \footcite{Lions1968} to optimize RCS by controling the skin current distribution: \cite{Bardos1995}
        \item 2001: Formulation of a BEM adjoint for optimization: \cite{Paoli2001}
        \item 2003: Use of Paoli's formulation for stealth optimization: \cite{Roge2003}
        \item 2011: Adjoint formulation for a Discontinuous Galerkin method for sensitivities computation: \cite{Wang2011}
        \item 2013: Discrete adjoint formulation with Ravier-Thomas base functions, validated on sensitivity analysis for an antenna: \cite{Kataja2013}
    \end{itemize}
\end{frame}

\subsection{Adjoint derivation}

\begin{frame}

    \frametitle{Continuous adjoint formulation}
    Let us rewrite the linear system of unknowns $\J$ defined by equation \eqref{eq:syslin} under the following form:
    \begin{equation}
        \mathcal R(\X(\eta),\J(\eta)) = 0,
        \label{eq:system-cont}
    \end{equation}
    and consider the observation function $\zeta(\X(\eta),\J(\eta))$.
    The right derivative of $\zeta$ with respect to $\eta$ writes:
    \begin{equation}
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} + \derpart{\zeta}{\J}\derpart{\J}{\eta}\right)\dd \eta.
        \label{eq:dsigmadx}
    \end{equation}

    \only<1>{
        $\X$: mesh coordinates

        $\eta$: optimization parameters
        }

    \onslide<2->
    The term $\derpart{\J}{\eta}$ is very costly to compute. However, it can be expressed in a more clever way since $\dd\RR = 0$:

    \begin{equation}
        \dd\RR = \left(\derpart\RR\X\derpart\X \eta + \derpart\RR\J\derpart\J \eta\right)\dd \eta = 0.
    \end{equation}
    \onslide<3->
    Then:
    \begin{equation}
        \derpart\J \eta = \left(\derpart\RR\J\right)^{-1}\derpart\RR\X\derpart\X \eta.
        \label{eq:djdx}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Continuous adjoint formulation II}
    Combining the system \eqref{eq:djdx} and the expression \eqref{eq:dsigmadx}, yields:

    \begin{equation}
        \boxed{
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} - \Psi^*\derpart{\mathcal R}{\X}\derpart{\X}{\eta}\right)\dd\eta.
        }
        \label{eq:dsigmadx-adj}
    \end{equation}

    Where $\Psi$ is the solution to the adjoint system: 
    \begin{equation}
        \left(\derpart{\mathcal R}{\J}\right)^* \Psi = \left(\derpart{\zeta}{\J}\right)^*.
        \label{eq:adjoint-cont}
    \end{equation}
\end{frame}


\begin{frame}
    \label{fra:smembre}
    \frametitle{Computation of the right-hand side}

    Let %\revised{us define an observable $\zeta$} 
    \begin{equation}
        \zeta = 2\sqrt{\pi}\left(\E^\infty\cdot\Einc\right),
    \end{equation}

    so that, the RCS can be written as $\sigma = 4\pi\frac{|\E^\infty|^2}{|\E\inc|^2} = |\zeta|^2$.

    \pause
    If $\uu_s$ is a unit vector and $\uu_s = \uu_i$, then
    \begin{equation}
        \E^\infty\cdot\Einc = \frac{kZ_0^2}{2\sqrt{\pi}}\bb.
    \end{equation}

    \pause
    Therefore, the differential of $\zeta$ is:
    \begin{equation}
        \dd\zeta = \frac{kZ_0^2}{2\sqrt{\pi}}\dd\bb.
    \end{equation}

    \pause
    Hence, the right-hand side of the discrete adjoint system can be written as:
    \begin{equation}
        \derpart{\zeta}{I} = \frac{kZ_0^2}{2\sqrt{\pi}}b.
    \end{equation}

    \pause
    From this, there is no need to actually solve the adjoint system and $\Psi$ is:
    \begin{equation}
        \Psi^* = \frac{kZ_0^2}{2\sqrt{\pi}}I.
    \end{equation}

\end{frame}
