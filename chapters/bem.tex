\subsection{The Boundary Element Method}
\begin{frame}
    \frametitle{The Electric Field Integral Equation}
    It can be shown\footcite{Nedelec1973} that, under the assumption of slide \ref{fra:hyp}, the solution \eqref{bem-eq:sol-helmholtz} admits the following integral representation :
    \begin{equation}
        \E(x) = E\inc(x) + jkZ_0\left(\frac1{k^2}\nabla_\Gamma\nabla_\Gamma\cdot+1\right)\int_\Gamma G(x,y)\J(y)\dd\Gamma(y)
    \end{equation}
    where $$G(x,y) = \frac{e^{-jk|x-y|}}{|x-y|}$$ is the three-dimension homogeneous Green function.
\end{frame}
\begin{frame}
    \frametitle{The Electric Field Integral Equation}
    The previous formula is more conveniently expressed in integral form. 
    This yields the following variational problem \footcite{Nedelec1973} :%\footnote<2->{Bendali, A.~, 2013, \textit{\'Equations Intégrales en \'Electromagnétisme}.} :

    \begin{equation}
    \boxed{\boxed{
    \left\{\begin{aligned}
        &\text{Trouver }\J\in\Hdiv(\Gamma), \forall\J'\in\Hdiv(\Gamma),& \\
        \int_\Gamma\int_\Gamma G(\rr,\rr') & \left(-\frac1{k^2}\nabla_\Gamma\cdot\J(\rr')\nabla_\Gamma\cdot\J'(\rr)+\J(\rr')\cdot\J'(\rr)\right)\dd\rr'\dd\rr &\\&=-\frac1{j\omega\mu_0}\int_\Gamma\E\inc\cdot\J'\dd\rr&
        \end{aligned}\right.
        \label{eq:efie}
    }}.
    \end{equation}

    Where thus introduce the space : $$\Hdiv(\Gamma) := \left\{\J\in L^2(\Gamma;\mathbb C^3),\J\text{ tangent to }\Gamma, \nabla_\Gamma\cdot\J\in L^2(\Gamma)\right\}.$$
\end{frame}


\begin{frame}
    \frametitle{Discretization}
        The induced surface current $\J$ can be expanded as a weighted sum of $N$ basis functions $\fn{\rr}$, with a set of weights $(I_n)_{1\leq n\leq N}\in\mathbb C^N$ :
    \begin{equation*}
        \J\approx\sum_{n=1}^N I_n\fn{\rr}.
    \end{equation*}
    \pause
    We will use Rao-Wilson-Glisson (RWG) basis functions defined as follows :
    \begin{equation*}
        \fn{\rr} = \left\{\begin{array}{cl}
            \frac{l_n}{2S_n^+}\left(\rr-\rr_n^+\right)&\rr\in T_n^+,\\
            \frac{l_n}{2S_n^-}\left(\rr_n^--\rr\right)&\rr\in T_n^-,\\
            0&\text{otherwise}.
        \end{array}\right.
    \end{equation*}
    $\rr_n^\pm$, $l_n$ and $T_n^\pm$ are defined on the figure bellow. $S_n^\pm$ is the area of the triangle $T_n^\pm$.

    \begin{figure}[ht]
        \centering
        \scalebox{.6}{\subimport{resources/bem}{def-RWG.tex}}
        \caption*{Sketch of the Rao-Wilson-Glisson finite elements.}
        \label{fig:rwg}
    \end{figure}

\end{frame}

\subsection{The BEM adjoint}
\begin{frame}
    \frametitle{Discrete adjoint formulation}

    Let $\mathcal R$ be a linear system such that, $$\mathcal R(\X(\eta),I(\eta)) = A(X(\eta))I(\eta) - \bb(X(\eta)) = 0,$$
    where $\X$ is the vector of mesh coordinates, $I$ the vector of unknown current weight, and $\eta$ the vector of design parameters.

    \pause
    Let $\zeta$ be an observable depending on $\X$ and $I$.

    \onslide<3->
    The derivative of $\zeta$ with respect to $\eta$ writes :
    \begin{equation}
        \dd{\zeta} = \derpart{\zeta}{\X}\derpart{X}{\eta}\dd\eta - \Psi^*\left({\color<3->{lightgray}\derpart{A}\X I }- \derpart{b}\X\right)\derpart{\X}{\eta}\dd\eta
    \end{equation}

    
    \onslide<4->
    where $\Psi$ is the solution of the adjoint system : :
    \onslide<5->
    \begin{equation}
        \overline{A}\Psi = \left(\derpart{\zeta}{I}\right)^*
    \end{equation}

    \onslide<6>
    Because of time constraints, we did not have the time to implement the computation (especially because of memory requirements and the handling of singular cases) of the term $\derpart{A}\X\J$, and so we omit it in the following.

\end{frame}

    

\subsection{The optimization}
% \subsection{Cost and constraints}


\begin{frame}%[label=fra:cost-bidis]
    \label{fra:cost-bem}
    \frametitle{Cost and constraints}
    Cost :
    \begin{equation*}
        F_e = \frac{\tilde\sigma}{{\tilde\sigma}^{ref}},
    \end{equation*}
    \onslide<2->
    where :
    \begin{equation*}
        \tilde\sigma = \int_{\theta_0}^{\theta_1}\int_{\phi_0}^{\phi_1} \Gamma_d(\theta)\delta_s(\sigma_\db)\sin(\phi)\dd\theta\dd\phi.
    \end{equation*}
    \onslide<3->
    \begin{itemize}[label=$\bullet$]
        \item $\delta_s$ : threshold function on RCS, to cut oscillations in low RCS zones.
        \item \hyperlink<3->{fra:gamma}{$\Gamma_d$} : \hypertarget<3->{lk:gamma-bem}{directivity function, to focus on critical directions.}
    \end{itemize}
    
    \onslide<4->
    Constraint :
    \begin{itemize}[label=$\bullet$]
        \item \sref{} constant
    \end{itemize}

    \onslide<5->
    RCS computation:
    \begin{itemize}[label=$\bullet$]
        \item 1 frequency, $f=500\unit{\mega\hertz}$
    \end{itemize}
    
\end{frame}
