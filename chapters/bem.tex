\section{The BEM optimization}

\subsection{The Boundary Element Method}
\begin{frame}
    \frametitle{The Electric Field Integral Equation}
    It can be shown\footnote<1->{Bendali, A.~, 2013, \textit{Integral Equations in Electromagnetics}.} that, under the assumption of slide \ref{fra:hyp}, the solution of equation \eqref{bem-eq:Helmholtz} admits the following integral representation :
    \begin{equation}
        \E(x) = E\inc(x) + jkZ_0\left(\frac1{k^2}\nabla_\Gamma\nabla_\Gamma\cdot+1\right)\int_\Gamma G(x,y)\J(y)\dd\Gamma(y)
    \end{equation}
    where $$G(x,y) = \frac{e^{-jk|x-y|}}{|x-y|}$$ is the three-dimension homogeneous Green function.
\end{frame}
\begin{frame}
    \frametitle{The Electric Field Integral Equation}
    The previous formula is more conveniently expressed in integral form. We thus introduce the space : $$\Hdiv(\Gamma) := \left\{\J\in L^2(\Gamma;\mathbb C^3),\J\text{ tangent to }\Gamma, \nabla_\Gamma\cdot\J\in L^2(\Gamma)\right\}.$$
    \onslide<2->
    This yields the follozing variational problem\footnote<2->{Bendali, A.~, 2013, \textit{\'Equations Intégrales en \'Electromagnétisme}.} :

    \begin{equation}
    \boxed{\boxed{
    \left\{\begin{aligned}
        &\text{Trouver }\J\in\Hdiv(\Gamma), \forall\J'\in\Hdiv(\Gamma),& \\
        \int_\Gamma\int_\Gamma G(\rr,\rr') & \left(-\frac1{k^2}\nabla_\Gamma\cdot\J(\rr')\nabla_\Gamma\cdot\J'(\rr)+\J(\rr')\cdot\J'(\rr)\right)\dd\rr'\dd\rr &\\&=-\frac1{j\omega\mu_0}\int_\Gamma\E\inc\cdot\J'\dd\rr&
        \end{aligned}\right.
        \label{eq:efie}
    }}.
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Discretization}
    The induced surface current $\J$ can be expanded as a weighted sum of $N$ basis functions $\fn{\rr}$, with a set of weights $(I_n)_{1\leq n\leq N}\in\mathbb C^N$ :
    \begin{equation*}
        \J\approx\sum_{n=1}^N I_n\fn{\rr}.
    \end{equation*}

\end{frame}

\begin{frame}
    \frametitle{Discretization}
    We will use Rao-Wilson-Glisson (RWG) basis functions defined as follows :
    \begin{equation*}
        \fn{\rr} = \left\{\begin{array}{cl}
            \frac{l_n}{2S_n^+}\left(\rr-\rr_n^+\right)&\rr\in T_n^+,\\
            \frac{l_n}{2S_n^-}\left(\rr_n^--\rr\right)&\rr\in T_n^-,\\
            0&\text{autrement}.
        \end{array}\right.
    \end{equation*}
    $\rr_n^\pm$, $l_n$ et $T_n^\pm$ are defined on the figure bellow. $S_n^\pm$ is the area of the triangle $T_n^\pm$.

    \begin{figure}[ht]
        \centering
        \scalebox{.6}{\subimport{resources/bem}{def-RWG.tex}}
        \caption*{Sketch of the Rao-Wilson-Glisson finite elements.}
        \label{fig:rwg}
    \end{figure}

\end{frame}

\begin{frame}
    \frametitle{Discretization}
    This allows to express the EFIE problem \eqref{eq:efie} under the form of a linear system :
    \begin{equation}
        A\J = \bb.
        \label{eq:syslin}
    \end{equation}
\end{frame}

\subsection{The BEM adjoint}
\begin{frame}

    \frametitle{Continuous adjoint formulation}
    Let us rewrite the linear system of unknowns $\J$ defined by equation \eqref{eq:syslin} under the following form :
    \begin{equation}
        \mathcal R(\X(\eta),\J(\eta)) = 0,
        \label{eq:system-cont}
    \end{equation}
    and consider the observation function $\zeta(\X(\eta),\J(\eta))$.
    The right derivative of $\zeta$ with respect to $\eta$ writes :
    \begin{equation}
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} + \derpart{\zeta}{\J}\derpart{\J}{\eta}\right)\dd \eta.
        \label{eq:dsigmadx}
    \end{equation}

    \only<1>{
        $\X$ : mesh coordinates

        $\eta$ : optimization parameters
        }

    \onslide<2->
    The term $\derpart{\J}{\eta}$ is very costly to compute. However, it can be expressed in a more clever way since $\dd\RR = 0$ :

    \begin{equation}
        \dd\RR = \left(\derpart\RR\X\derpart\X \eta + \derpart\RR\J\derpart\J \eta\right)\dd \eta = 0.
    \end{equation}
    \onslide<3->
    Then :
    \begin{equation}
        \derpart\J \eta = \left(\derpart\RR\J\right)^{-1}\derpart\RR\X\derpart\X \eta.
        \label{eq:djdx}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Continuous adjoint formulation II}
    Injecting the system \eqref{eq:djdx} into the expression \eqref{eq:dsigmadx}, yields :

    \begin{equation}
        \boxed{
        \dd{\zeta} = \left(\derpart{\zeta}{\X}\derpart{\X}{\eta} - \Psi^*\derpart{\mathcal R}{\X}\derpart{\X}{\eta}\right)\dd\eta.
        }
        \label{eq:dsigmadx-adj}
    \end{equation}

    Where $\Psi$ is the solution to the adjoint system: 
    \begin{equation}
        \left(\derpart{\mathcal R}{\J}\right)^* \Psi = \left(\derpart{\zeta}{\J}\right)^*.
        \label{eq:adjoint-cont}
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Discrete adjoint formulation}

    With a discrete formulation, $$\mathcal R(\X,\J) = A(X)\J - \bb(X).$$

    \onslide<2->
    The derivative of $\zeta$ with respect to $\eta$ writes :
    \begin{equation}
        \dd{\zeta} = \derpart{\zeta}{\X}\dd{X} - \Psi^*\left({\color<3->{lightgray}\derpart{A}\X }- \derpart{b}\X\right)\dd{\X}
    \end{equation}

    \only<3>{
    Since the geometry variations are small, we assume that $\derpart{A}{\X}$ is negligible compared to the other terms.
    }

    \onslide<4->
    \only<5->{In the EFIE formulation, $A$ is non-Hermitian symmetric i.e.~ $A^* = \overline{A}$.}

    the adjoint system thus writes :
    \only<4>{
    \begin{equation}
        A^*\Psi = \left(\derpart{\zeta}{\J}\right)^*
    \end{equation}}
    \onslide<5->
    \begin{equation}
        \overline{A}\Psi = \left(\derpart{\zeta}{\J}\right)^*
    \end{equation}

\end{frame}

\begin{frame}
    \label{fra:smembre}
    \frametitle{Computation of the right-hand side}

    Let %\revised{us define an observable $\zeta$} 
    \begin{equation}
        \zeta = 2\sqrt{\pi}\left(\E^\infty\cdot\Einc\right),
    \end{equation}

    so that, the RCS can be written as $\sigma = 4\pi\frac{|\E^\infty|^2}{|\E\inc|^2} = |\zeta|^2$.

    \pause
    If $\uu_s$ is a unit vector and $\uu_s = \uu_i$, then
    \begin{equation}
        \E^\infty\cdot\Einc = \frac{kZ_0^2}{2\sqrt{\pi}}\ell(\J).
    \end{equation}

    \pause
    Therefore, the differential of $\zeta$ is :
    \begin{equation}
        \dd\zeta = \frac{kZ_0^2}{2\sqrt{\pi}}\ell(\dd\J).
    \end{equation}

    \pause
    Hence, the right-hand side of the discrete adjoint system can be written as :
    \begin{equation}
        \derpart{\zeta}{I} = \frac{kZ_0^2}{2\sqrt{\pi}}b.
    \end{equation}

    \pause
    From this, there is no need to actually solve the adjoint system and $\Psi$ is :
    \begin{equation}
        \Psi^* = \frac{kZ_0^2}{2\sqrt{\pi}}I.
    \end{equation}

\end{frame}
