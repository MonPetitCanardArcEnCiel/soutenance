\subsection{The Physical Optics}
\begin{frame}
    \frametitle{The Physical Optics}
    Let us recall the far field expression \eqref{eq:champ-lointain} :
    
    \begin{equation}
        \E^\infty(\tfrac\x{|\x|}) = \frac{kZ_0}{4\pi}\frac\x{|\x|}\times\left(\mathcal F(\J)(\tfrac{\x}{|\x|})\times\frac\x{|\x|}\right).
        \label{eq:champ-lointain-2}
    \end{equation}
    
    where :
    \begin{equation}
        \mathcal F(\J)(\x) = \int_\Gamma\J(y)e^{-jk\x\cdot y}\dd\Gamma(y).
    \end{equation}
\end{frame}
\begin{frame}
    \frametitle{The Physical Optics}
    \begin{block}{Hypothesis}
        The frequency is high enough that the wavelength is small compared to the characteristic dimensions of the object.
    \end{block}
    \only<2>
    {
        At high frequencies ($\sim1\unit{\giga\hertz}$ and more), this hypothesis is reasonable : for $f = 2\unit{\giga\hertz}$, $\lambda = 15\unit{\centi\meter}$ where a fighter aircraft's characteristic lenghts $\sim10\unit{meter}$.
    }
    % \pause
    \onslide<3->
    The Physical Optics approximation, the current distribution $\J$ induced by a plane wave $\E\inc$ propagating along the direction $\uu_i$ is given by\footnote<2->{Chatzigeorgiadis, F.~, 2004, \textit{Development of Code for a Physical Optics RADAR Cross Section Prediction and Analysis Application}.} :
    \begin{columns}
        \begin{column}{0.75\textwidth}
            \begin{equation}
                \J(y) = \left\{\begin{array}{cr}
                    \left(\dfrac2{Z_0}\mathbf n\times\mathbf \uu_i\times\E\inc\right)e^{jk\uu_i\cdot y}&\text{in the illumated zone,}\\
                    \mathbf 0&\text{in the shadow zone,}
                \end{array}\right.
            \end{equation}
        \end{column}
        \begin{column}{0.25\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{images/bidis/sketch_shadowzone.png}
                % \caption*{Illustration of the illuminated and shadow zones.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{The Physical Optics Far Field}
    Finally, the far field scattered electric field writes :
    \begin{equation}
        \E^\infty(\uu_s) = \frac{2k}{4\pi}\int_\Gamma\uu_s\times\left(\n\times \uu_i\times\E\inc\right)\times\uu_s e^{jk(\uu_i-\uu_s)\cdot y}\dd\Gamma(y)
    \end{equation}
\end{frame}

\begin{frame}
    \frametitle{Gradient validation}
    \only<1-2>{
    Consider $f$, a real-valued function, at least 3 times differentiable and $f'$ its derivative. Let $h>0$. Then $\forall x$ :
    \begin{align*}
        \frac{f(x+h)-f(x)}{h} = f'(x) +  \textcolor<2>{red}{o(h)}\quad\text{as }h\to0,\\
        \frac{f(x+h)-f(x-h)}{2h} = f'(x) +  \textcolor<2>{red}{o(h^2)}\quad\text{as }h\to0.
    \end{align*}
    \onslide<2>
    \vspace{\baselineskip}
    \textcolor{red}{Truncation error}
    }
    \onslide<3->
    {
    \begin{figure}
        \centering
        \captionsetup{justification=centering}
        \begin{subfigure}
            {0.49\linewidth}
            \includegraphics[width=\textwidth, height=.75\textheight, keepaspectratio]{images/bidis/valid_grad_ex2.png}
            % \caption*{Finite differences}
        \end{subfigure}
        \begin{subfigure}
            {0.49\linewidth}
            \includegraphics[width=\textwidth, height=.75\textheight, keepaspectratio]{images/bidis/valid_grad_ex.png}
            % \caption*{Finite differences}
        \end{subfigure}
        \caption*{Exemple of error between the exact gradient and the finite differences (Rosenbrock function)}
    \end{figure}
    }

    \begin{itemize}[label=\ding{227}]
        \item The error between FD and the exact gradient should decrease until a critical $\epsilon$, where numerical errors become predominant.
        \item The slope of the error log should be 1 for forward FD and 2 for centered FD (near the critical $\epsilon$).
    \end{itemize}

\end{frame}

\subsection{The optimization}
\begin{frame}%[label=fra:cost-bidis]
    \label{fra:cost-bidis}
    \frametitle{Cost and constraints}
    Cost :
    \begin{center}
        $F = wF_a + (1-w)F_e,$	with $0\leq w\leq1$.
    \end{center}

    \begin{columns}
        \begin{column}<2->{.5\textwidth}
            \begin{center}Stealth criterion\end{center}
            \begin{equation*}
                F_e = \frac{\tilde\sigma}{{\tilde\sigma}^{ref}},
                \label{eq:Fe}
            \end{equation*}
            \onslide<3->
            \begin{equation*}
                \tilde\sigma = \int_{\theta_0}^{\theta_1}\int_{\phi_0}^{\phi_1} \Gamma_d(\theta)\delta_s(\sigma_\db)\sin(\phi)\dd\theta\dd\phi.
                \label{eq:sigma-tilde}
            \end{equation*}
            \onslide<4->
            \begin{itemize}[label=$\bullet$]
                \item $\delta_s$ : threshold function on RCS, to cut oscillations in low RCS zones.
                \item \hyperlink<4->{fra:gamma}{$\Gamma_d$} : directivity function, to focus on critical directions.
            \end{itemize}
        \end{column}

        \begin{column}<5->{.5\textwidth}
            \begin{center}Aerodynamic criterion\end{center}
                \begin{equation*}
                    F_a = \frac{C_D}{{C_D}^{ref}}.
                    \label{eq:Fa}
                \end{equation*}
                \onslide<6->
                \begin{center}Aerodynamic constraints\end{center}
            \begin{itemize}[label=$\bullet$]
                \item \sref{} constant
                \item \cl{} constant
            \end{itemize}

        \end{column}
    \end{columns}
    
\end{frame}
