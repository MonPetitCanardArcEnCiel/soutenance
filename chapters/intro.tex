\begin{frame}
    \frametitle{Optimization}

    The problem:

    For a given parameter space $\mathbb P$ (ex: wing sweep angles, chord length, span, camber...):
    \begin{align*}
        \underset{X\in\mathbb P}{\argmin} \left\{\begin{array}{l}
            \text{aerodynamic criterion}(X)\\
            \text{stealth criterion}(X)
        \end{array}\right.\\
        \text{s.~t.~}\quad\left\{\begin{array}{l}
            \text{aerodynamic constraints}\\
            \text{stealth constraints}\\
            \text{geometric constraints}
        \end{array}\right.
    \end{align*}
    with $X$ the vector of geometric parameters.
\end{frame}


\begin{frame}
    \frametitle{Tradeoffs}

\end{frame}

\begin{frame}
    \frametitle{Optimization algorithms}
    
    \begin{columns}
        \column{.5\textwidth}
            \textbf{Gradient-free algorithms}
            \begin{itemize}[label=\ding{227}]
                \item<2-> Require a lot of function evaluations
                \item<4-> Computation cost rises quickly with number of parameters.
                \item<5-> No gradient
                \item<6-> Tend to struggle with constraints\footnotemark
                % \item<6-> Some algorithms are global
            \end{itemize}
        \column{.5\textwidth}
            \textbf{Gradient-based algorithms}
            \tikzmark{start}
            \begin{itemize}[label=\ding{227}]
                \item<2-> Require less function evaluations
                \item<4-> Computation cost less dependant on number of parameters.
                \item<5-> Require an efficient way to compute the gradient
                % \item<6-> Local
            \end{itemize}
    \end{columns}

\subsection{State of the art}
\begin{frame}
    \frametitle{State of the art -- Aero-stealth optimization I}
    \small
    Few studies have used integrated aero-stealth optimization methods.
    \begin{itemize}[label=\ding{227},leftmargin=5mm]
        \item Until the late 2010's, most of them used gradient-free algorithms:
        \begin{itemize}[label=\ding{229}]
            \item<2-> 2007: Evolutionary algorithm with a potential flow solver and PO for planform and wing section optimization (very low fidelity methods): \cite{Lee2007}
            % \item 2014: Adaptative surrogate method with a BLISS formulation for structure, stealth and aerodynamic optimization of high aspect ratio wing \cite{Wu2014}
            \item<3-> 2016: Surrogate-based optimization with RANS and Finite Volume Time Domain method for aero-stealth optimization of a helicopter rotor (2 parameters): \cite{Jiang2016}
            % \item<4-> 2018: Multidisciplinary optimization of drones for radar signature, sensor performance and trajectory: \cite{Papageorgiou2018}
            \item<4-> 2020: Multi-objective EGO\footcite{Guerra2016} to optimize the aero-stealth of a UCAV with RANS and Geometrical Optics (2 parameters): \cite{Garnier2020}
            % \item<8-> 2021: Hierarchic kriging for aero-stealth optimization of flying wing:\pause \cite{Liu2021}
            % \item<4-> 2023: Gaussian Process to optimize  aero-stealth of a fighter aircraft: \cite{Taj2023}
        \end{itemize}
    \end{itemize}
\end{frame}


\begin{frame}
    \frametitle{State of the art -- Aero-stealth optimization II}
    \small
    Few studies have used integrated aero-stealth optimization methods.
    \begin{itemize}[leftmargin=5mm]
        \item Since the late 2010's, some gradient-based frameworks emerge:
        \begin{itemize}[label=\ding{227}]
            % \item 2017: Gradient-based aero-stealth optimization of a flying wing, with analytical gradient: \cite{Pan2017}
            \item 2019: Constrained gradient-based aero-stealth optimization of a UCAV: \cite{Li2019}
            \item 2020: Use of BEM adjoint and RANS for aero-stealth high fidelity optimization of the wing profiles of a flying wing geometry. The adjoint terms are computed by finite differences. \cite{Zhou2020}
        \onslide<3-> 
            \begin{itemize}[label=\ding{229}]
                \item<2-> Academic use-case
                \item<2-> CAD free deformation $\implies$ difficult to include in an industrial framework
                \item<2-> Simplistic stealth criterion (average RCS)
            \end{itemize}
        \end{itemize}
    \end{itemize}
\end{frame}
