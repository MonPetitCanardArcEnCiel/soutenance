\section{From the maxwell equations to the RCS}

\subsection{Fundamentals}
\begin{frame}
    \frametitle{The Maxwell Equations}


    Let us recall the Maxwell equations in the harmonic regime :
    \begin{subequations}
	\begin{equation}
		\nabla \cdot{} \D = \varrho, \label{bem-eq:divD}
	\end{equation}
	\begin{equation}
		\nabla \cdot \B = 0, \label{bem-eq:divB}
	\end{equation}
	\begin{equation}
		\nabla \times \E = j\omega\B, \label{bem-eq:rotE}
	\end{equation}
	\begin{equation}
		\nabla \times \Hh = \J -j\omega\D, \label{bem-eq:rotH}
	\end{equation}
        % \pause
        along with the continuity equation : 
        \begin{equation}
            \label{eq:cont}
            \nabla\cdot\J + \derpartt{\rho} = 0.
        \end{equation}
    \end{subequations}
    \onslide<1->
    \small
    \begin{equation*}
        \begin{array}{ccc}
            \E\text{ : electric field}&\J\text{ : current density}\\
            \B\text{ : magnetic field}&\rho\text{ : charge density}\\
            \D = \varepsilon_0\E&\varepsilon_0\text{ : dielectric permitivity}\\
            \B = \mu_0\Hh&\mu_0\text{ : magnetic permeability}\\
            k = \dfrac{2\pi f}{c}&Z_0 = \sqrt{\dfrac{\mu_0}{\varepsilon_0}}
        \end{array}
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{The Vector Helmholtz Equation}
    \begin{equation}
    -\left(\nabla^2+k^2\right)\E = \mathbf T    
    \label{bem-eq:Helmholtz}
\end{equation}

With : 
\begin{equation}
    \mathbf T = kZ_0\left(\frac1{k^2}\nabla\nabla\cdot+1\right)\J.
\end{equation}

\pause
The solution to equation \eqref{bem-eq:Helmholtz} can be expressed as :

\begin{equation}
    \E(x) = \int_\Omega \mathbf T(y)G(x,y)\dd\Omega(y),\quad \forall x\notin\mathrm{supp}(\mathbf T),
    \label{bem-eq:sol-helmholtz}
\end{equation}
 

    where $$G(x,y) = \frac{e^{-jk|x-y|}}{|x-y|}$$ is the three-dimension homogeneous Green function.
\end{frame}


\subsection{Far field and RCS}

\begin{frame}
    \frametitle{The Far Field}
    \label{fra:hyp}


    \begin{columns}
        \begin{column}{0.75\textwidth}
            \begin{block}<1->{Assumption}
                Assume an ideal perfectly conducting obstacle occupying a bounded domain $\Omega^-$ with boundary $\Gamma$ in free space $\Omega$ and illuminated by an incident plane wave $\E\inc$ propagating along the direction $\uu_i$.
            \end{block}
        \end{column}
        \begin{column}{0.25\textwidth}
            \begin{figure}
                \centering
                \captionsetup{justification=centering}
                \includegraphics[width=\textwidth]{images/rcs/sketch_target.png}
                % \caption*{Sketch of the obstacle illuminated by the plane wave.}
                \label{fig:scatterer}
            \end{figure}
        \end{column}
    \end{columns}

    \onslide<2->
    It can be shown that, in the absence of losses, the solution \eqref{bem-eq:sol-helmholtz} can be expressed as :
    \begin{equation}
        \E(x) = j\frac{e^{ik|x|}}{|x|}\left\{\E^\infty(\tfrac x{|x|}) + \mathcal{O}\left(\frac1{|x|}\right)\right\},
        \label{eq:helmholtz}
    \end{equation}
    %
    \onslide<2->
    with
    \begin{equation}
        \E^\infty(\tfrac\x{|\x|}) = \frac{kZ_0}{4\pi}\frac\x{|\x|}\times\left(\mathcal F(\J)(\tfrac{\x}{|\x|})\times\frac\x{|\x|}\right),
        \label{eq:champ-lointain}
    \end{equation}
    %
    \onslide<3->
    where
    \begin{equation}
        \mathcal F(\J)(\x) = \left<\J_y,e^{-jkx\cdot y}\right>_{\varepsilon'(\mathbb R^3),\varepsilon(\mathbb R^3)}.
        \label{eq:cond-F}
    \end{equation}

    \onslide<4->
    The quantity $\E^\infty$ is called the far field.

\end{frame}

\begin{frame}
    \frametitle{The Radar Cross Section (RCS)}
    The Radar Cross Section (RCS) of an object in the direction ($\uu_s$) is defined as :
    \begin{equation}
    \label{bem-eq:RCSdef}
    \sigma(\uu_s)= \lim\limits_{R\to\infty} R^2\frac{||\E(R\uu_s)||^2}{||\E_i||^2},
\end{equation}
%
\pause

From the development \eqref{eq:champ-lointain}, the limit in \eqref{bem-eq:RCSdef} is finite and the RCS is well defined and can be expressed as :

    
\begin{equation}
    \sigma(\uu_s) =  4\pi\norm{\E^\infty(\uu_s)}^2.
\end{equation}

\pause

To compute $\sigma$, one needs to compute the current density $\J$ on the scatterer surface.

\end{frame}
